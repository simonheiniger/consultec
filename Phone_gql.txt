// createInvoice()
//=================================================================================================
sub createInvoice
  (
  aJsonIn : handle; // cte
  var tJsonOut : handle; // mem
  ) : logic;
local
{
  tRet : int;
  tJson : handle;
  tCode : alpha(4096);
  tMem : handle;
  tFind : int;
  tQrInvoice : logic;
  tBR : alpha(2);
  tCodeMem : handle;
}
{
  tCode # Lib_Json:ReadAlpha(aJsonIn,'createInvoice/code');
  
  //--- Stadard br einf?gen --------------------------
  tBR # StrChar(13)+StrChar(10);
  tCode # Lib_Str:rep(tCode,tBR,StrChar(10),true);
  tCode # Lib_Str:rep(tCode,StrChar(10),tBR,true);
  
  //--- Validation File --------------------------------
  if('d'+'bg' = 'live')
  {
    tCodeMem # Lib_Mem:alphaToMem(tCode)
    Lib_Mem:Convert(tCodeMem,_sys->spCharset,_CharsetUTF8)
    
    Lib_File:MemToFile(tCodeMem,'C:/C16/QR.txt')
  }
 
  //--- Check Code -----------------------------------
  if(StrCut(tCode,1,3) =^ 'SPC')
    tQrInvoice # true
  
  //--- schon vorhanden -----------------------------
  if(tQrInvoice)
  {
    Lib_Text:TextToMem('QR',var tMem)
    tFind # MemFindStr(tMem,1,tMem->spLen,tCode,_StrCaseIgnore)
    MemFree(tMem);

    if(tFind = 0)
      tRet # Lib_Text:AlphaToText(tCode,'QR');
  }
  
  
  switch(true)
  {
    case !tQrInvoice : tCode # 'WRONG';
    case tFind > 0 : tCode # 'EXISTING';
    case tRet = 0 : tCode # 'OK';
    default : tCode # 'FALSE'
  }
  
  
  
  tJsonOut # Lib_Mem:AlphaToMem('{"data":{"createInvoice":{');
  Lib_Mem:AlphaAppend(tJsonOut,'"code":"'+tCode+'"');
  Lib_Mem:AlphaAppend(tJsonOut,'}}}');

  return(true);
}

sub getOrder
  (
  aJsonIn : handle; // cte
  var tJsonOut : handle; // mem
  ) : logic;
local
{
  tRet : int;
  tJson : handle;
  tOrderNummer : int;
  qOrder : handle;
  qOrderItem : handle;
  qFahrzeug : handle;
  qAdresse : handle;
  tItemCount : int;
}
{
  tOrderNummer # Lib_Json:ReadInt(aJsonIn,'getOrder/orderNumber');
  qOrder # RecBufCreate(150);
  qOrderItem # RecBufCreate(962);
  qFahrzeug # RecBufCreate(30);
  qAdresse # RecBufCreate(1);
  
  qOrder->"WAU-DatIdt" # tOrderNummer
  tRet # RecRead(qOrder,1,0);
  
  if(tRet = 0)
  {
    tRet # RecLink(qFahrzeug,qOrder,2,_RecFirst);
    if(tRet!= 0)
      RecBufClear(qFahrzeug);
    tRet # RecLink(qAdresse,qOrder,1,_RecFirst);
    if(tRet!= 0)
      RecBufClear(qAdresse);
    
    tJsonOut # Lib_Mem:AlphaToMem('{"data":{"getOrder":{');
      Lib_Mem:AlphaAppend(tJsonOut,'"orderNumber":"'+CnvAI(qOrder->"WAU-DatIdt",_FmtNumNoGroup)+'",');
      Lib_Mem:AlphaAppend(tJsonOut,'"modell":"'+qFahrzeug->"FAZ-Fahrzeugtyp"+'",');
      Lib_Mem:AlphaAppend(tJsonOut,'"marke":"'+qFahrzeug->"FAZ-Marke"+'",');
      Lib_Mem:AlphaAppend(tJsonOut,'"kundeVorname":"'+qAdresse->"ADR-Vorname"+'",');
      Lib_Mem:AlphaAppend(tJsonOut,'"kundeNachname":"'+qAdresse->"ADR-Name"+'",');
      Lib_Mem:AlphaAppend(tJsonOut,'"items":[');
        for tRet # RecLink(qOrderItem,qOrder,12,_RecLast);
        loop tRet # RecLink(qOrderItem,qOrder,12,_RecPrev);
        while(tRet = 0)
        {
          Inc(tItemCount);
          Lib_Mem:AlphaAppend(tJsonOut,'{');
          Lib_Mem:AlphaAppend(tJsonOut,'"id":'+CnvAI(qOrderItem->"WAL-Position",_FmtNumNoGroup)+',');
          Lib_Mem:AlphaAppend(tJsonOut,'"artikelNummer":"'+qOrderItem->"WAL-Artikelnummer"+'",');
          Lib_Mem:AlphaAppend(tJsonOut,'"name":"'+qOrderItem->"WAL-Bezeichnung"+'",');
          Lib_Mem:AlphaAppend(tJsonOut,'"quantity":'+CnvAF(qOrderItem->"WAL-Anzahl",_FmtNumNoGroup | _FmtNumPoint,0,2));
          Lib_Mem:AlphaAppend(tJsonOut,'},');
        }
        if(tItemCount > 0)
          tJsonOut->spLen # tJsonOut->spLen-1;
        
      Lib_Mem:AlphaAppend(tJsonOut,']');
    Lib_Mem:AlphaAppend(tJsonOut,'}}}');
  }
  else
  {
    tJsonOut # Lib_Mem:AlphaToMem('{"data":{"getOrder":{}}}');
  }
  
  Lib_Mem:Convert(tJsonOut,_sys->spCharset,_CharsetUTF8);
  
  
  RecBufDestroy(qOrder);
  RecBufDestroy(qOrderItem);
  
  return(true);
}

//=================================================================================================
// lagBarcode() - Empfängt EAN-Nummer und schreibt sie als String in GV-Barcode
//                ueber Web_Service:setGlobal. Bei Fehler Fallback in Textdatei 'EAN'.
//=================================================================================================
sub lagBarcode
  (
  aJsonIn      : handle;
  var tJsonOut : handle;
  ) : logic;
local
{
  tRet      : int;
  tEan      : alpha(80);
  tEanInt   : int;
  tSuccess  : logic;
  tFallback : logic;
  tMessage  : alpha(255);
}
{
  // Lese EAN-Nummer aus JSON-Input
  tEan # StrCut(Lib_Json:ReadAlpha(aJsonIn,'lagBarcode/ean'),1,80);
  tEan # StrAdj(tEan, _StrAll);

  tFallback # false;

  // Konvertiere EAN-String zu int fuer GV-Barcode
  tEanInt # CnvIA(tEan);

  // Primaer: EAN als int in globale Variable GV-Barcode schreiben
  tSuccess # Web_Service:setGlobal('GV-Barcode', tEanInt);

  if(tSuccess)
  {
    tMessage # 'EAN erfolgreich in GV-Barcode gespeichert';
  }
  else
  {
    // Fallback: in Textdatei 'EAN' schreiben (alte Logik)
    tRet # Lib_Text:AlphaToText(tEan,'EAN');
    if(tRet = 0)
    {
      tSuccess  # true;
      tFallback # true;
      tMessage  # 'GV-Barcode fehlgeschlagen - EAN in Textdatei gespeichert (Fallback)';
    }
    else
    {
      tMessage # 'Fehler: weder GV-Barcode noch Textdatei beschreibbar';
    }
  }

  // JSON-Antwort
  tJsonOut # Lib_Mem:AlphaToMem('{"data":{"lagBarcode":{');
  Lib_Mem:AlphaAppend(tJsonOut,'"ean":"'+tEan+'",');

  if(tSuccess)
    Lib_Mem:AlphaAppend(tJsonOut,'"success":true,');
  else
    Lib_Mem:AlphaAppend(tJsonOut,'"success":false,');

  if(tFallback)
    Lib_Mem:AlphaAppend(tJsonOut,'"fallback":true,');
  else
    Lib_Mem:AlphaAppend(tJsonOut,'"fallback":false,');

  Lib_Mem:AlphaAppend(tJsonOut,'"message":"'+tMessage+'"');
  Lib_Mem:AlphaAppend(tJsonOut,'}}}');

  Lib_Mem:Convert(tJsonOut,_sys->spCharset,_CharsetUTF8);

  return(true);
}

// main()
//=================================================================================================
main
  (
  aQuery : alpha(4096);
  var tJsonOut : handle; // Mem
  ) : logic;
local
{
  tRet : int;
  tRef : alpha;
  tJsonIn : handle; // Cte
  tJson : alpha(4096);
  tSchemaName : Alpha;
  tSchema : handle;
}
{
aQuery # Lib_Str:Rep(aQuery,'__26__','&'); //stefan nicht gut

Lib_Json:AlphaToJson(aQuery,var tJsonIn);
if(tJsonIn <= 0)
{
  Lib_Json:AlphaToMemUtf8('{"error":"Json Parse Error"}',var tJsonOut);
  return(false);
}

tSchema # Lib_Json:ReadNode(tJsonIn,'');
if(tSchema <= 0)
{
  Lib_Json:AlphaToMemUtf8('{"error":"Json Parse Error"}',var tJsonOut);
  return(false);
}


switch(tSchema->spName)
{
  //---- Global ---------------------------------
  case 'getPref' : getPref(tJsonIn,var tJsonOut);
  
  //--- Order ----------------------------------
