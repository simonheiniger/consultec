// __Convert()
//-------------------------------------------------------------------------------------------------
sub __Convert
(
  aMem : handle;
  aCharsetSource : int;
  aCharsetTarget : int;
)
local
{
  tTarget : handle;
  tLen : int;
  tPosBegin : int;
  tPosEnd : int;
  tPos : int;
}
{
  tTarget # MemAllocate(_MemAutoSize);
  aMem->spCharset # aCharsetSource;
  MemCnv(aMem,tTarget,aCharsetTarget);

  MemWriteByte(aMem,1,0,aMem->spLen);
  aMem->spLen # 0;
  MemCopy(tTarget,1,tTarget->spLen,1,aMem);
  tTarget->MemFree();
}

// __FileToMem()
//-------------------------------------------------------------------------------------------------
sub __FileToMem
(
  aFilePNE : alpha(200);
  var aMem : handle;
  opt aCharsetFile : int;
): logic;
local
{
  tRet : int;
  tFsi : handle;
}
{
  if(aCharsetFile = 0)
   aCharsetFile #  _CharsetUTF8;
   
   
  tFsi # FsiOpen(aFilePNE,_FsiAcsR);
  if(tFsi <= 0)
    return(false);

  aMem # MemAllocate(FsiSize(tFsi)+100);
  tRet # FsiReadMem(tFsi,aMem,1,FsiSize(tFsi));
  FsiClose(tFsi);
  
  __Convert(aMem,aCharsetFile ,_sys->spCharset)

  if(tRet > 0)
    return(true);
  else
    return(false);
}


//-------------------------------------------------------------------------------------------------
// __StrRep
//-------------------------------------------------------------------------------------------------
sub __StrRep
(
  aStr : alpha(4096);
  aRep : alpha(100);
  aIns : alpha(100);
  opt aAll : logic;
) : alpha;
local
{
  tStart : int;
  tLen : int;
  tFind : int;
  tInsLen : int;
}
{
  tInsLen # StrLen(aIns);
  tStart # 1;
  for tFind # StrFind(aStr,aRep,tStart,_StrCaseIgnore)
  loop tFind # StrFind(aStr,aRep,tStart,_StrCaseIgnore)
  while(tFind != 0)
    {
    aStr # StrDel(aStr,tFind,StrLen(aRep));
    aStr # StrIns(aStr,aIns,tFind);
    if (aAll = false)
      break;
    tStart # tFind+tInsLen;
    }
  return(aStr);
}


//-------------------------------------------------------------------------------------------------
// __goToPath
//-------------------------------------------------------------------------------------------------
sub __goToPath
(
  aJson : handle;
  aPath : alpha;
): handle;
local
{
  tFind : int;
  tStart : int;
  tEnd : int;
  tPath : alpha(4096);
  tToken : alpha;
  tIndex : int;
  tItem : handle;
  tCount : int;
  tItem2 : handle;
}
{
  if(aJson <= 0)
    return(0);
    
    
  if(aPath = '')
  {
    tItem # aJson->CteRead(_CteChildList | _CteFirst);
    return(tItem);
  }
  
  tItem # aJson;
  aPath # __StrRep(aPath,'.','/',true);
  
  tStart # 1;
  for tFind # StrFindRegEx(aPath,'[\[\(]',tStart)
  loop tFind # StrFindRegEx(aPath,'[\[\(]',tStart)
  while(tFind > 0)
  {
    tPath # StrCut(aPath,tStart,tFind-tStart)
    tItem # tItem->CteRead(_CteNodePath, 0,tPath);
    if(tItem <= 0)
      return(0);
    
    tEnd # StrFindRegEx(aPath,'[\]\)]',tStart)
    
    
    //--- Index --------------------------
    tIndex # CnvIA(StrCut(aPath,tFind+1,tEnd-tFind-1))
    tCount # 1;
    for tItem2 # tItem->CteRead(_CteFirst);
    loop tItem2 # tItem->CteRead(_CteNext,tItem2);
    while(tItem2 > 0 and tCount < tIndex)
    {
      Inc(tCount);
    }
    tItem # tItem2;
    tStart # tEnd+1;
  }
  if(tItem <= 0)
      return(0);

  tPath # StrCut(aPath,tStart,StrLen(aPath)-tStart+1);
  if(tPath != '')
    tItem # tItem->CteRead(_CteNodePath, 0,tPath);
  
  return(tItem);
}
