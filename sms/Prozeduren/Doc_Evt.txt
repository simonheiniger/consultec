//=================================================================================================
//  EvtClose()
//=================================================================================================
sub EvtClose
  (
  aEvt              : event;    // Ereignis
  ): logic;
local
{
  tFile : alpha(500);
  tRet : int;
  tCloseApp : logic;
}
{
if(aEvt:Obj->wpName = 'Doc_Main')
{
  if (gSaved)
  {
    gSaveAction # 1;
    return(true);
  }
  else
  {
    if(gNewVersion)
    {
      Doc_Main:Save(gCtx,gAction,gTableNo,gDatId,gFileSrcPNE,gEmlAccId,Mdl_Lib:GetCh($chNotify),gView,gDocName,var gSaveAction, var tCloseApp);
    }
    else
    {
      Doc_Main:SaveDialogOld(gCtx,gAction,gTableNo,gDatId,gFileSrcPNE,gEmlAccId,Mdl_Lib:GetCh($chNotify),gView,var gSaveAction, var tCloseApp);
    }
    return(tCloseApp);
  }
}
}

sub EvtClicked
  (
  aEvt              : event;    // Event
  ): logic;
local
{
  tPos            : int;
  tHdl            : handle;
  tResult         : int;
  tInputPos       : handle;
}
{

switch (aEvt:Obj->wpName)
{
  case 'btnGo' :
  {
    gCtx->cpiSelLength # 0;
    gCtx->cpiSelStart # gCtx->cpiGetCharFromLine($edLine->wpCaptionInt - 1);
    gCtx->WinFocusSet();
    gCtx->cphCurrentInputPosition # gCtx->cphCurrentInputPosition;
  }
  case 'btnUp' : Doc_Ctx:SearchText(gCtx, $edSearch, Doc_Ctx:GetState($cbMatchCase), txFindSearchUp); // Suchrichtung oben
  case 'btnDown' : Doc_Ctx:SearchText(gCtx, $edSearch, Doc_Ctx:GetState($cbMatchCase)); // Suchrichtung unten
  case 'btnwbarClose' : $wbarBottom->wpVisible # false; // Suchbar ausblenden
  case 'btnReplaceOne' : Doc_Ctx:ReplaceOne(); // ausgewählten ersetzen
  case 'btnReplaceAll' : Doc_Ctx:ReplaceAll(); // Alle ersetzen
  case 'chNotify' : Doc_Main:SetSaveStatus('NotSaved');
  case 'btBold' : Doc_ToolBox:SetBold();
  case 'btItalic' : Doc_ToolBox:SetItalic();
  case 'btUnderline' : Doc_ToolBox:SetUnderline();
  case 'btStrikethru' : Doc_ToolBox:SetStrikethru();
  case 'btPrintSerial' : Doc_PrintMulti();
}
return(true);
}

//=================================================================================================
sub EvtMenuCommand
  (
  aEvt              : event;    // Ereignis
  aMenuItem         : handle;   // Auslösender Menüpunkt / Toolbar-Button
  ): logic;
local
{
  tRet : int;
  tLogic : logic;
  tDate : Date;
  tAlpha : alpha;
  aEmlPFE : alpha(512);
  aDocPFE : alpha(512);
  tFile : alpha(512);
  tYear : alpha(4);
  tFilePNE : alpha(512);
  tCloseApp : logic;
}
{
aEmlPFE # Doc_Lib:GetFilePNE(gTableNo,gDatId);
aDocPFE # Doc_Lib:GetFilePNE(gTableNo,gDatId);

if(aMenuItem->wpName =* 'miInsertTemplate*')
{
  tFile # StrCut(aMenuItem->wpName,21,15);
  tYear # StrCut(aMenuItem->wpName,17,4);
  Doc_ToolBox:InsertTemplate(gCtx,Lib_Sys:DbP()+'Emails\'+tYear+'\'+FsiSplitName(tFile,_FsiNameN)+'\'+tFile);
}
switch (aMenuItem->wpName)
{
  //--- Speichern -------------------------------
  case 'miSave' :
{
    if(gNewVersion)
    {
      Doc_Lib:DocSave(gCtx,gAction,gTableNo,gDatId,gEmlAccId,Mdl_Lib:GetCh($chNotify),gView);
      gFrmMain->WinClose();
    }
    else
  {
      Doc_Lib:DocSaveOld(gCtx,gAction,gTableNo,gDatId,gFileSrcPNE,gEmlAccId,Mdl_Lib:GetCh($chNotify),gView);
      gFrmMain->WinClose();
    }
  }
  case 'miEmlTemplate' : Doc_TemplateSave($Doc_Main,gAction,gAccParentId,$edSubject->wpCaption,gDocName,gView);
  //--- Export ----------------------------------
  case 'miExportPdf' : Eml_Export(gFrmMain,gCtx,0,'',gDocName,'pdf',false,true);
  case 'miExportDoc' : Eml_Export(gFrmMain,gCtx,0,'',gDocName,'doc',false,true);
  case 'miExportDocX' : Eml_Export(gFrmMain,gCtx,0,'',gDocName,'docx',false,true);
  case 'miExportTxt' : Eml_Export(gFrmMain,gCtx,0,'',gDocName,'txt',false,true);
  case 'miExportEml' : Eml_Export(gFrmMain,gCtx,0,'',gDocName,'eml',false,true);
  case 'miExportHtml' : Eml_Export(gFrmMain,gCtx,0,'',gDocName,'html',false,true);
  case 'miExportRtf' : Eml_Export(gFrmMain,gCtx,0,'',gDocName,'rtf',false,true);

  //--- Drucken ---------------------------------
  case 'miSetSite' :
{
    tLogic # gCtx->cplSectionFormatDialog(0);
    Doc_Main:SetSaveStatus('NotSaved');
  }
  case 'miPrint' : Doc_Main:Print(gFrmMain,gCtx,gSerialFunction,gView,true);// Druckt
  case 'miExit', 'tbnExit' : gFrmMain->WinClose();
  //--- E-Mail ----------------------------------
  case 'miAttAdd' : Doc_Att:AttAdd(); // Datei anfügen
  case 'miViewModeEmlSingle2' : Doc_Main:View('EmlSingle',gSerialFunction);

  //--- Bearbeiten ------------------------------
  case 'miUndo' :
{
    tLogic # gCtx->cplUndo;
    // Falls Aktion fehlschlägt oder es nichts zum Rückgängig machen gibt => Menüpunkt disablen
    if (gCtx->cpiCanUndo = 0 OR !tLogic)
      $miUndo->wpDisabled # true;
    if (gCtx->cpiCanRedo != 0)
      $miRedo->wpDisabled # false;
  }
  case 'miRedo' :
  {
    tLogic # gCtx->cplRedo;
    // Falls Aktion fehlschlägt oder es nichts zum Wiederholen gibt => Menüpunkt disablen
    if (gCtx->cpiCanRedo = 0 OR !tLogic)
      $miRedo->wpDisabled # true;
    if (gCtx->cpiCanUndo != 0)
      $miUndo->wpDisabled # false;
  }
  case 'miCut' : Doc_Ctx:Cut(gCtx);
  case 'miCopy' : Doc_Ctx:Copy(gCtx);
  case 'miPaste' : Doc_Ctx:Paste(gCtx);
  case 'miSearch' : Doc_Ctx:EnableSearchbar('Search');
  case 'miReplace' : Doc_Ctx:EnableSearchbar('Replace');
  case 'miGoTo' : Doc_Ctx:EnableSearchbar('GoTo');


  //--- Link ------------------------------------
  case 'miLinkOpen' : Doc_Ctx:LinkOpen(gCtx);
  case 'miLinkEdit' : Doc_Link(gCtx);
  case 'miLinkRemove' : Doc_Ctx:LinkRemove(gCtx);


  //--- Ansicht ---------------------------------
  case 'miViewEml' : Doc_Main:View('EmlSingle',gSerialFunction);
  case 'miViewDoc' : Doc_Main:View('Doc',gSerialFunction);
  case 'miSerialFunction' :
  {
    switch(gView)
    {
      case 0 : Doc_Main:View('Doc',!gSerialFunction);
      case 1 : Doc_Main:View('EmlSingle',!gSerialFunction);
      case 3 : Doc_Main:View('Html',!gSerialFunction);
    }


  }
  case 'miViewHtml' : Doc_HtmlPreview();
  case 'miGrids','miGrids2' : // Gitternetzlinien anzeigen/ausblenden
  {
    tLogic # $miGrids->wpMenuCheck;
    $miGrids->wpMenuCheck # !tLogic;
    gCtx->cplTableGridLines # !tLogic;
    Eml_Pref:UpdateLogic(gPref,'Pref/DocGridlines',gCtx->cplTableGridLines);
  }
  //--- Extras ----------------------------------
  case 'miContact' : Eml_Contact(0);
  case 'miSpellCheck' :
  {
    aMenuItem->wpMenuCheck # !gCtx->cplEnableSpellChecking;
    gCtx->cplEnableSpellChecking # !gCtx->cplEnableSpellChecking;
  }
  case 'miEnglishUS' : Doc_Ctx:SetDictionary(_WinDictEnglishUS, aMenuItem);
  case 'miEnglishUK' : Doc_Ctx:SetDictionary(_WinDictEnglishUK, aMenuItem);
  case 'miEnglishCA' : Doc_Ctx:SetDictionary(_WinDictEnglishCA, aMenuItem);
  case 'miEnglishTech' : Doc_Ctx:SetDictionary(_WinDictEnglishTech, aMenuItem);
  case 'miGerman' : Doc_Ctx:SetDictionary(_WinDictGermanDE, aMenuItem);
  case 'miPrefDialog' : Eml_PrefDialog(gFrmMain,gPref,gUserId);
  //--- Schliessen ------------------------------
  case 'miShowHelp' : Doc_Main:Help();
  case 'miInfo' : Doc_Info();
}
return(true);
}

//=================================================================================================
//  EvtCtxEvent()
//=================================================================================================
sub EvtCtxEvent
  (
  aEvt              : event;    // Ereignis
  aEventID          : int;      // Ereignis-Nummer
  aComArguments     : handle;   // COM-Argument-Objekt
  ): logic;
local
{
  tLink           : alpha(4096);
  tKeyCode        : int;
  tShift          : int;
  tLogic          : logic;
  tErr            : int;
  tErrText        : alpha(512);
  tObjectID       : int;
  tTmp            : alpha(4096);
  tCtxDocEdit     : handle;
  tField : int;
  tInt1 : int;
  tInt2 : int;
  tInt3 : int;
  tInt4 : int;
  tRet : int;
  tCloseApp : logic;
}
{
//bgTracE('aEventID: '+CnvAI(aEventID));
//--- Events ------------------------
switch (aEventID);
{
  case txEvtMouseUp :
  {
    aComArguments->ComArgGet(1,tInt1); // Button 1=Left 2=Rigth 3=Middle
    aComArguments->ComArgGet(2,tInt2); // Shinft 1=Shift 2=CTRL 3=Alt
    aComArguments->ComArgGet(3,tInt3); // Pos X
    aComArguments->ComArgGet(4,tInt4); // Pos Y
    if(tInt1 = 2)
    {
      //--- Set Cursor --------------------------
      tRet # ComCall(gCtx,'InputPosFromPoint',Doc_Lib:PxToTwips(tInt3),Doc_Lib:PxToTwips(tInt4));
      if(tRet != -1)
      {
        if(!(tRet >= gCtx->cpiSelStart and tRet <= gCtx->cpiSelStart+gCtx->cpiSelLength))
        {
          gCtx->cpiSelStart # tRet;
          gCtx->cpiSelLength # 0;
        }
      }
      //--- Context Menue -----------------------
      if(gView = 0) // Email
        WinMenuContext(gCtx,'Doc_Buttons',tInt3+24,tInt4+24);
      else
        WinMenuContext(gCtx,'Doc_Buttons',tInt3+0,tInt4+0);
    }

  }
  case txEvtObjectDblClicked :
  {
    tLogic # gCtx->cplObjectAttrDialog;
    if (tLogic = true)
      Doc_Main:SetSaveStatus('NotSaved');
  }
  case txEvtChange :
  {
    Doc_Main:SetSaveStatus('NotSaved');
  }
  case txEvtPosChange :
  {
    Doc_ToolBox:InitToolBox();

    Doc_Ctx:SetChanges();
  }
  case txEvtError :
  {
    aComArguments->ComArgSet(7,true);
    aComArguments->ComArgGet(1,tErr);
    aComArguments->ComArgGet(2,tErrText);
    try
    {
      ErrSet(tErr);
    }
    //Doc_Error:WriteLog('Code: '+CnvAI(tErr)+' | '+tErrText,false);  zuviele E-Mails
    WinDialogBox(gFrmMain,'SwissText',tErrText, _WinIcoError, _WinDialogOk, 0);
  }
  case txEvtFieldDblClicked :
  {
    aComArguments->ComArgGet(1, tField);
    gCtx->cpiFieldCurrent # tField;
    if(gCtx->cpiFieldType(tField) = txFieldExternalLink);
    {
      Doc_Ctx:LinkOpen(gCtx);
      //tLink # gCtx->cpaFieldTypeData(tField)
      //SysExecute('cmd', '/c start '+FsiSplitName(tLink,_FsiNameNE), 0);
    }
  }
  case txEvtKeyDown :
  {
    aComArguments->ComArgGet(1, tKeyCode);
    aComArguments->ComArgGet(2, tShift);
    if (tKeyCode = 77 and tShift = 2)
      gShortKey # 1; // Ctrl + M abfangen
    if (tKeyCode = 77 and tShift = 3)
      gShortKey # 1; // Shift + Ctrl + M abfangen
    if (tKeyCode = 78 and tShift = 2)
      gShortKey # 1; // Ctrl + N abfangen
    if (tKeyCode = 78 and tShift = 3)
      gShortKey # 1; // Shift + Ctrl + N abfangen
  }
  case txEvtKeyPress :
  {
  if (gShortKey != 0)
    {
    aComArguments->ComArgSet(1,0);
    gShortKey # 0;
    }
  aComArguments->ComArgGet(1, tKeyCode);
  // Esc
  if (tKeyCode = 27)
    {
    if (WinDialogBox(gFrmMain,'SwissMail','Möchten Sie das Fenster wirklich schliessen?',_WinIcoQuestion,_WinDialogYesNo,_WinIdNo) = _WinIdYes)
      gFrmMain->WinClose();
    }
  }
  case txEvtKeyUp :
  {
    aComArguments->ComArgGet(1, tKeyCode);
    aComArguments->ComArgGet(2, tShift);
    // Steuerung wurde gedrückt
    if (tShift = 2)
    {
      switch(tKeyCode)
      {
        // Strg + E
        case 69 : Doc_HtmlPreview();
        // Strg + F
        case 70 : Doc_Ctx:EnableSearchbar('Search');
        // Strg + G
        //case 71 : Doc_Ctx:EnableSearchbar('GoTo');
        // Strg + H
        case 72 : Doc_Ctx:EnableSearchbar('Replace');
        // Strg + M
        case 77 :
        {
          switch(true)
          {
            case $miViewEml->wpMenuCheck : Doc_Main:View('Doc',gSerialFunction);
            case $miViewDoc->wpMenuCheck : Doc_Main:View('EmlSingle',gSerialFunction);
          }
        }

        // Strg + P
        case 80 : Doc_Main:Print(gFrmMain,gCtx,gSerialFunction,gView,true);
        // Strg + R
        case 82 :
        {
          $miSpellCheck->wpMenuCheck # !gCtx->cplEnableSpellChecking;
          gCtx->cplEnableSpellChecking # !gCtx->cplEnableSpellChecking;
        }
        // Strg + S
        case 83 :
        {
          if (gSaveMenuDisabled = false)
          {
            if(gNewVersion)
            {
              if(gAction = 'DocNew')
                Doc_Main:InsertDialog(gCtx,gAction,gTableNo,gDatId,gFileSrcPNE,gEmlAccId,Mdl_Lib:GetCh($chNotify),gView,gDocName,var gSaveAction, var tCloseApp);
              else
              {
                Doc_Lib:DocSave(gCtx,gAction,gTableNo,gDatId,gEmlAccId,Mdl_Lib:GetCh($chNotify),gView);
                // gFrmMain->WinClose();
              }
            }
            else
            {
              Doc_Lib:DocSaveOld(gCtx,gAction,gTableNo,gDatId,gFileSrcPNE,gEmlAccId,Mdl_Lib:GetCh($chNotify),gView);
              gFrmMain->WinClose();
            }
          }
        }
      }
    }
  }
}
return(true);
}

//=================================================================================================
sub EvtMenuInitPopup
  (
  aEvt                 : event;    // Ereignis
  aMenuItem            : handle;   // Auslösender Menüeintrag
  ): logic;
local
{
  tAccId : int;
  tFilter : handle;
  qTemplate : handle;
  tRet : int;
  tCount : int;
}
{
switch(aMenuItem->wpName)
{

  case 'miFileHead' :
  {
    //--- Speichern ausblenden ------------------
    if (gSaveMenuDisabled = true)
      $miSave->wpDisabled # true;
  }
  case 'miTable' :
  {
    //--- Tabelle -------------------------------
    $miTableProp->wpDisabled # !gCtx->cplTableCanChangeAttr;
    $miInsertColRight->wpDisabled # !gCtx->cplTableCanChangeAttr;
    $miInsertRowDown->wpDisabled # !gCtx->cplTableCanChangeAttr;
    $miDeleteCol->wpDisabled # !gCtx->cplTableCanChangeAttr;
    $miDeleteRow->wpDisabled # !gCtx->cplTableCanChangeAttr;
    $miTableSplit->wpDisabled # !gCtx->cplTableCanChangeAttr;
  }
  case 'miInsert' :
  {
    if("Sys-Mail")
    {
      $miTemplate->WinMenuItemRemove();
      $miInsert->WinMenuItemAdd('miTemplate', 'Vorlagen');
      //--- Filter --------------------------------
      qTemplate # RecBufCreate(511);
      tAccId # Eml_Lib:GetFolder(gAccParentId,'temp');
      tFilter # RecFilterCreate(511,2);
      tFilter->RecFilterAdd(7,_FltAND,_FltScan,Eml_Lib:IntToFilter(tAccId));
      tCount # 0;
      for tRet # RecRead(qTemplate,2,_RecFirst,tFilter);
      loop tRet # RecRead(qTemplate,2,_RecNext,tFilter);
      while(tRet = 0)
      {
        Inc(tCount);
        $miTemplate->WinMenuItemAdd('miInsertTemplate'+sCnvAI(qTemplate->EHI.CreateDate->vpYear)+qTemplate->EHI.File,qTemplate->EHI.Subject);
      }
      if(tCount = 0)
        $miTemplate->wpDisabled # true;
      qTemplate->RecBufDestroy();
      tFilter->RecFilterDestroy();
    }
    else
    {
      $miTemplate->WinMenuItemRemove();
      $miInsert->WinMenuItemAdd('miTemplate', 'Vorlagen [Option von SwissMail]');
      $miTemplate->wpDisabled # true;
    }
  }
}
return(true);
}
