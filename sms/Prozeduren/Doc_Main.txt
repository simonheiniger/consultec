// SetGlobal()
//=================================================================================================
sub SetGlobal
  (
  aAction : alpha(16);
  aTableNo : int;
  aDatId : int;
  aAlpha1 : alpha(150);
  aAlpha2 : alpha(150);
  aSaveMenuDisabled : logic;
  aView : int;
  aEmlAccId : int;
  aIsClone : logic;
  aDocName : alpha(100);
  aReplaceVariables : logic;
  aSerialFunction : logic;
  aTempFolder : alpha(500);
  aNewVersion : logic;
  aTemplateName : alpha(500);
  ): int;
local
  {
  tCalTime : CalTime;
  tYear : int;
  tMonth : int;
  qEml : handle;
  tRet : int;
  }
{
VarAllocate(DocGlobal);
tCalTime->vmServerTime();
tYear # tCalTime->vpYear;
tMonth # tCalTime->vpMonth;
//--- Systemdaten lesen -------------------------
RecRead(997,1,_RecFirst);

//--- Lizenz ------------------------------------
gLicenseNumber # StrCut(DbaLicense(_DbaSrvLicense),1,10);
// gLicenseNumber # 'CS153650WN'; // parfin de siecle

//--- Speicherort E-Mail Archiv -----------------
FsiPathCreate(Lib_Sys:DbP()+'Emails');
FsiPathCreate(Lib_Sys:DbP()+'Emails\'+CnvAI(tYear,_FmtNumNoGroup));
if (tMonth = 12)
  FsiPathCreate(Lib_Sys:DbP()+'Emails\'+CnvAI(tYear+1,_FmtNumNoGroup));

//--- Speicherort Dokument Archiv ---------------
FsiPathCreate(Lib_Sys:DbP()+'Documents');
FsiPathCreate(Lib_Sys:DbP()+'Documents\'+CnvAI(tYear,_FmtNumNoGroup));
if (tMonth = 12)
  FsiPathCreate(Lib_Sys:DbP()+'Documents\'+CnvAI(tYear+1,_FmtNumNoGroup));

//--- Temp Path ---------------------------------
if(aTempFolder = '');
gTempPath # Mdl_Lib:CreateTempFolder('DocMain');


//--- Action ------------------------------------
gSaved # true;
gTableNo # aTableNo;
gDatId # aDatId;
switch(true)
  {
  case (aAction =^ 'opendoc') : gAction # 'DocOpen';
  case (aAction =^ 'newdoc') : gAction # 'DocNew';
  case (aAction =^ 'copybin') : return(-1);
  case (aAction =^ 'delbin') : gAction # 'delbin';
  default : gAction # aAction;
  }

//--- Attributes ---------------------------------
if(aAlpha2 = '') // neue Methode
  {
  if(aAlpha1 != '')
    gFileSrcPNE # Lib_Sys:DbP()+'Documents\'+FsiSplitName(aAlpha1,_FsiNamePN)+'\'+FsiSplitName(aAlpha1,_FsiNameN)+'.fsx';
  }
else // alte Metode
  {
  switch(aAlpha1)
    {
    case 'TempUser','Templates' : aAlpha1 # Lib_Sys:DbP()+'Documents\Templates';
    }
  if(StrCut(aAlpha1,StrLen(aAlpha1),1) != '\')
    aAlpha1 # aAlpha1+'\';

  gFileSrcPNE # aAlpha1+FsiSplitName(aAlpha2,_FsiNamePN)+'\'+FsiSplitName(aAlpha2,_FsiNameN)+'.fsx';
  }


//-----------------------------------------------

gSaveMenuDisabled # aSaveMenuDisabled;
gIsClone # aIsClone;
gBinDirTempSys # BinDirOpen(0, 'TempSys', _BinCreate);
gPrintDataListType# StrCnv("SEB-Art",_StrLower);
if(gPrintDataListType = '')
  gPrintDataListType # 'adr';
gMailDataListType # StrCnv("SEB-Art",_StrLower);
if(gMailDataListType = '')
  gMailDataListType # 'adr';
gView # aView;
gEmlAccId # aEmlAccId;
gAccId # aEmlAccId;
gAccParentId # Eml_Lib:GetParent(aEmlAccId);
gUserCurrent # CnvIA(UserInfo(_UserCurrent));
gUserId # Doc_Lib:GetUserNameCurrent();
gReplaceVariables # aReplaceVariables;
gSerialFunction # aSerialFunction;
gNewVersion # aNewVersion;
gTemplateName #aTemplateName;

gDocName # aAlpha2;
if (StrFind(gAction,'doc',1,_StrCaseIgnore) != 0)
  {
  gDocName # aDocName;
  }

return(0);
}

// ================================================================================================
//   SetSaveStatus()
// ================================================================================================
sub SetSaveStatus
  (
  aStatus : alpha(10);
  )
{
//--- Speichern deaktivieren ---
if (gSaveMenuDisabled = true)
  aStatus # 'Off';

switch (aStatus)
  {
  case 'New' :
    {
    //$sbnText->wpImageTile # _WinImgNew;
    $sbnText->wpHelpTip # 'Datei wurde neu angelegt.'
    $miSave->wpDisabled # false;
    gSaved # false;
    }
  case 'Saved' :
    {
    //$sbnText->wpImageTile # _WinImgSave;
    $sbnText->wpHelpTip # 'Datei ist in dieser Form gespeichert.'
    $miSave->wpDisabled # true;
    gSaved # true;
    }
  case 'NotSaved' :
    {
    //$sbnText->wpImageTile # _WinImgClear;
    $sbnText->wpHelpTip # 'Datei ist in dieser Form noch nicht gespeichert.'
    $miSave->wpDisabled # false;
    gSaved # false;
    }
  case 'Off' :
    {
    //$sbnText->wpImageTile # _WinImgSave;
    $sbnText->wpHelpTip # 'Temporäre Datei'
    $miSave->wpDisabled # true;
    gSaved # true;
    }
  }
}

//======================================================================================================
//  Logfile schreiben
//======================================================================================================
sub WriteLog
  (
  aAction : Alpha(16); // mail,email,mailmulti,emailmulti
  aAddId : int;
  )
local
  {
  tErr : int;
  tBufText : handle;
  tBufAdr : handle;
  tAdrIdAlpha : alpha(16);
  tTextName : alpha(32);
  tDate : date;
  tDateAlpha : alpha(16);
  tLine : alpha(255);
  }
{
tBufAdr # RecBufCreate(1);
tBufAdr->"ADR-DatIdt" # aAddId;
tErr # RecRead(tBufAdr,1,0);
if (tErr != 0)
  return;
tBufText # TextOpen(512);
tAdrIdAlpha # CnvAI(aAddId,_FmtNumNoGroup);
tTextName # 'AKOM-' + tAdrIdAlpha;
tDate->vmServerTime();
tDateAlpha # CnvAD(tDate, _FmtInternal);

if (tBufAdr->"ADR-KommText" = N)
  {
  RecRead(tBufAdr,1,_RecLock);
  tBufAdr->"ADR-KommText" # Y;
  tBufAdr->"ADR-KomText" # tTextName;
  RecReplace(tBufAdr,_RecUnLock);
  }
RecBufDestroy(tBufAdr);
tErr # TextCreate(tTextName,0);
tErr # tBufText->TextRead(tTextName,0);
switch(aAction)
  {
  case 'mail' :
    {
    if ("KOM-ZHD DatIdt" = 0)
      tLine # 'Brief vom '+tDateAlpha+' '+gDocName;
    else
      tLine # 'Brief vom '+tDateAlpha+' '+gDocName+ ' '+"ZHD-Name"+','+ "ZHD-Vorname";
    }
  case 'email' :
    {
    if ("KOM-ZHD DatIdt" = 0)
      tLine # 'E-Mail vom '+tDateAlpha+' '+$edSubject->wpCaption;
    else
      tLine # 'E-Mail vom '+tDateAlpha+' '+$edSubject->wpCaption+ ' '+"ZHD-Name"+','+ "ZHD-Vorname";
    }
  case 'mailmulti' :
    {
    if ("KOM-ZHD DatIdt" = 0)
      tLine # 'Serienbrief vom '+tDateAlpha+' '+gDocName;
    else
      tLine # 'Serienbrief vom '+tDateAlpha+' '+gDocName+ ' '+"ZHD-Name"+','+ "ZHD-Vorname";
    }
  case 'emailmulti' :
    {
    if ("KOM-ZHD DatIdt" = 0)
      tLine # 'Serien E-Mail vom '+tDateAlpha+' '+gDocName;
    else
      tLine # 'Serien E-Mail vom '+tDateAlpha+' '+gDocName+ ' '+"ZHD-Name"+','+ "ZHD-Vorname";
    }
   
  case 'sms' :
    {
    if ("KOM-ZHD DatIdt" = 0)
      tLine # 'SMS vom '+tDateAlpha+' '+gDocName;
    else
      tLine # 'SMS vom '+tDateAlpha+' '+gDocName+ ' '+"ZHD-Name"+','+ "ZHD-Vorname";
    }
  }
//--- Text schreiben ------------------------------
If (tBufText->TextSearch(1,1,_TextSearchCI,tLine)=0)
    tBufText->TextLineWrite(1,tLine,_TextLineInsert );

tErr # tBufText->TextWrite(tTextName,0);
tBufText->TextClose();
}

//================================================================================================
//   InsertDialog()
//================================================================================================
sub InsertDialog
  (
  aCtx : handle;
  aAction : alpha;
  aTableNo : int;
  aDatId : int;
  aFileSrcPNE : alpha(500);
  aEmlAccId : int;
  aNotify : logic;
  aView : int;
  aDocName : alpha(500);
  var aSaveAction : int;
  var tCloseApp : logic;
  ): logic;
local
  {
  tRet : int;
  tFrame : handle;
  }
{
tFrame # Lib_Frm:Open('Doc_SaveNew');
$edDocName->wpcaption # aDocName;

Mdl_Lib:SetCh($FileTypeText,false);
Mdl_Lib:SetCh($rbLetter,false);
Mdl_Lib:SetCh($rbSerialLetter,false);

switch(aTableNo)
  {
  case 130 : Mdl_Lib:SetCh($FileTypeText,true);
  case 140 : Mdl_Lib:SetCh($rbLetter,true);
  case 141 : Mdl_Lib:SetCh($rbSerialLetter,true);
  }
Lib_Frm:Run(tFrame,$Doc_Main);


aDocName # $edDocName->wpcaption;
switch(true)
  {
  case Mdl_Lib:GetCh($FileTypeText) : aTableNo # 130;
  case Mdl_Lib:GetCh($rbLetter) : aTableNo # 140;
  case Mdl_Lib:GetCh($rbSerialLetter) : aTableNo # 141;
  }


tRet # WinDialogResult(tFrame);
switch(tRet)
  {
  case 1 : // Yes
    {
    aSaveAction # 1;
    Doc_Lib:DocInsert(aCtx,aTableNo,aFileSrcPNE,aDocName,aNotify,aView);

    if (gSaved)
      tCloseApp # true;
    else
      tCloseApp # false;
    }
  case 0 : // No
    {
    aSaveAction # 0;
    tCloseApp # true;
    }
  case -1 : // cancel
    {
    aSaveAction # -1;
    tCloseApp # false;
    }
  }
WinClose(tFrame);
}

//--- DB Buffer ---------------------------------
RecBufCopy(qDoc,aTableNo);

//--- BlobName ----------------------------------
if(aSaveAction = 1)
  {
  Doc_Lib:SetBlobName(aTableNo,tBlobName);
  }
else
  {
  if(gAction = 'DocNew') // nicht gespeichert
    {
    Doc_Lib:SetBlobName(aTableNo,'');

    switch(aTableNo)
      {
      //case 100 : "FAK-Bezeichnung" # ''; // Rechnung FAK
      case 130 : "TXT-Bezeichnung" # ''; // Text TXT
      case 140 : "KOM-Bezeichnung" # ''; // Brief KOM
      case 141 : "SEB-Bezeichnung" # ''; // serien Brief SEB
      //case 150 : "WAU-Bezeichnung" # ''; // Werkstattauftrag WAU
      //case 160 : "BES-Bezeichnung" # ''; // Bestellung BES
      }
    }
  }


//--- View ------------------------------------------
if(aView = 1)
  tEmail #true;
switch(aTableNo)
    {
    //case 100 : "FAK-EMail" # tEmail; // Rechnung FAK
    case 130 : "TXT-EMail" # tEmail; // Text TXT
    case 140 : "KOM-EMail" # tEmail; // Brief KOM
    case 141 : "SEB-EMail" # tEmail; // serien Brief SEB
    //case 150 : "WAU-EMail" # tEmail; // Werkstattauftrag WAU
    //case 160 : "BES-EMail" # tEmail; // Bestellung BES
    }

//--- Notification ------------------------------
if(aSaveAction != 0)
  Doc_Lib:SetNotification(aTableNo,tNotification);

RecBufDestroy(qDoc);
}
