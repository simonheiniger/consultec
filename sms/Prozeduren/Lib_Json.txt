//-------------------------------------------------------------------------------------------------
// getRootNode()
//-------------------------------------------------------------------------------------------------
sub getRootNode
  (
  aCte : handle;
  aPath : alpha(500);
  var tRoot : handle;
  var tNodeName : alpha;
  ) : handle;
local
  {
  tNode : handle;
  tRet : int;
  tPath : alpha;
  tFind : int;
  tLen : int;
  }
{
  tLen # StrLen(aPath);
  tFind # StrFind(aPath,'/',1,_StrFindReverse);
  if(tFind = 0)
  {
    tRoot # aCte;
    tNodeName # aPath;
  }
  else
  {
    tNodeName # StrCut(aPath,tFind+1,tLen-tFind);
    tPath # StrCut(aPath,1,tFind-1);
    tRoot # aCte->CteRead(_CteNodePath, 0, tPath);
    if(tRoot = 0)
      return(-1);
  }

return(0);
}

  tNodeName : alpha;

 tRet # getRootNode(aCte,aPath,var tRoot,var tNodeName);

return(tNode);
}

//-------------------------------------------------------------------------------------------------
// TextToJson()
//-------------------------------------------------------------------------------------------------
sub TextToJson
(
  aTextName : alpha;
  var aCte : handle;
) : int;
local
{
  tRet : int;
  tMem : handle;
}
{
  tRet # Lib_Text:TextToMem(aTextName, var tMem);
  if(tRet != 0)
  {
    MemFree(tMem);
    return(-1);
  }
  aCte # CteOpen(_CteNode);
  aCte->spID # _JsonNodeObject;
  tRet # JsonLoad(aCte,'',0,tMem);
  if(tRet != 0)
  {
    MemFree(tMem);
    return(-2);
  }
  MemFree(tMem);
  return(0);
}

// InsertArray()
//-------------------------------------------------------------------------------------------------
sub InsertArray
  (
  aCte : handle;
  aPath : alpha(128);
  ) : int;
local
  {
  tItem : handle;
  tPath : alpha(128);
  tFind : int;
  tNode : alpha(64);
  }
{
tFind # StrFind(aPath,'/',1,_StrFindReverse);
if (tFind = 0)
  {
  tNode # aPath;
  tItem # aCte->CteInsertNode(tNode,_JsonNodeArray,NULL);
  }
else
  {
  tNode # Strcut(aPath,tFind+1,StrLen(aPath)-tFind);
  tPath # Strcut(aPath,1,tFind-1);
  tItem # aCte->CteRead(_CteNodePath, 0, tPath);
  if(tItem = 0)
    return(0);
  tItem # tItem->CteInsertNode(tNode,_JsonNodeObject,NULL);
  }
return(tItem);
}

//-------------------------------------------------------------------------------------------------
// endcodeContent()
//-------------------------------------------------------------------------------------------------
sub endcodeContent
(
  aContent : alpha(4096);
): alpha;
local
{
  tCount : int;
  tLen : int;
  tChar : int;
  tContent : alpha(4096)
}
{
  tLen # StrLen(aContent);
  for tCount # 1
  loop Inc(tCount)
  while(tCount <= tLen)
  {
    tChar # StrToChar(aContent,tCount);
    switch(true)
    {
      case tChar = 9  : tContent # tContent+'\t'; //Tab to be replaced with \t
      case tChar = 10 : tContent # tContent+'\n'; //Newline to be replaced with \n
      case tChar = 13 : tContent # tContent+'\r'; // Carriage return to be replaced with \r
      case tChar < 32 : cycle;                    // Backspace to be replaced with \b, Form feed to be replaced with \f
      case tChar = 34 : tContent # tContent+'\"'; // Double quote to be replaced with \"
      case tChar = 92 : tContent # tContent+'\\'; // Backslash to be replaced with \\
      default :  tContent # tContent+StrChar(tChar);
    }
  
  }
  return tContent;

}
